# 原理学习
在Windows系统下，软件实现自我卸载主要依赖于安装时创建的卸载入口和独立的卸载程序。

安装时在注册表中创建条目，路径为：
`HKEY_LOCAL_MACHINE\SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall\{YourApp}`

64位系统需额外处理Wow6432Node分支
```ini
"DisplayName" = "Your App Name"        // 显示名称
"UninstallString" = "C:\Path\Uninstall.exe /silent"  // 卸载命令
"InstallLocation" = "C:\YourApp\"       // 安装路径
```

本质上看来，卸载程序与安装程序密切相关啊。想要卸载程序还得先去学习安装程序，但是目前很多的安装程序都是有好看的界面的，借用win的合成觉得乏味，要是能自定义就好了。

   ```python
    def schedule_self_removal():
        """使用任务计划程序删除自身"""
        import tempfile
        import subprocess
        
        uninstaller_path = Path(sys.argv[0]).resolve()
        install_dir = uninstaller_path.parent
        
        # 创建VBS脚本
        vbs_content = f"""
    Set shell = CreateObject("WScript.Shell")
    Set fso = CreateObject("Scripting.FileSystemObject")

    ' 等待卸载程序退出
    WScript.Sleep 2000

    ' 删除安装目录
    On Error Resume Next
    fso.DeleteFolder "{install_dir}", True

    ' 删除VBS脚本自身
    fso.DeleteFile WScript.ScriptFullName
    """
        
        # 保存VBS脚本
        vbs_path = Path(tempfile.gettempdir()) / f"self_delete_{os.getpid()}.vbs"
        with open(vbs_path, "w") as f:
            f.write(vbs_content)
        
        # 执行VBS脚本
        subprocess.Popen(
            ['wscript', str(vbs_path)],
            creationflags=subprocess.CREATE_NO_WINDOW
        )
   ```

## bat删除

### 安全删除逻辑**
检查进程是否被关闭，最多检测30秒，然后强制杀死进程，开始执行删除程序
```bat
:: 关闭命令回显使输出更清晰。
@echo off
:: 允许在循环内使用变量延迟扩展
setlocal enabledelayedexpansion
set max_wait=30
set wait_count=0

:: 设置回调地点
:check_process

:: 设置杀死的进程名称
set process_name="爱丽丝QQ聊天AI.exe"
:: 检查进程是否仍在运行
:check_process
tasklist /fi "imagename eq %process_name%" | findstr /i "%process_name%"
if %errorlevel% equ 0 (
    if %wait_count% geq %max_wait% (
        :: 强制终止进程(进程正常结束或者是超过30秒到这里被强制结束)
        taskkill /f /im "%process_name%"
        :: 跳出回调
        exit /b 1
    ) else (
        :: 等待一秒
        ping -n 1000 127.0.0.1
        :: 退出时间加1
        set /a wait_count+=1
        :: 重新回调
        goto :check_process
    )
)
```
### 删除文件和目录
del (或 erase) - 删除文件

基本语法：
1. del 参数 "文件路径"
   1. /f: 强制删除只读文件。
   2. /q: 安静模式。删除时不要求确认（慎用！）。
   3. /s: 从当前目录及其所有子目录中删除指定文件。
   4. /p: 删除每个文件前提示确认。
2. rd (或 rmdir) - 删除目录（文件夹）
   1. /s: 删除目录树（即删除指定目录及其包含的所有子目录和文件）。
   2. /q: 安静模式。与 /s 一起使用时，删除整个目录树不需要确认（非常慎用！）。
   3. rd 本身只能删除空目录。要删除包含文件的目录，必须使用 /s 选项。

`*`匹配任意字符序列（零个或多个）。del *.tmp 删除当前目录所有 .tmp 文件。
`?`匹配单个任意字符。del report?.docx 删除 report1.docx, reportA.docx 等。


***使用ANSI编码，如果bat文件是UTF-8编码，会导致乱码，没法删除中文目录文件***
其实也能整VBS脚本文件
```bat
:: 关闭命令回显使输出更清晰。
@echo off
:: 文件夹删除(删除目录及其所有文件、删除时不要求确认)

rd /s /q "文档"

:: exe程序卸载(强制删除只读文件、删除时不要求确认)

```

```python
def uninstall_program():
    """卸载程序"""
    # 获取当前可执行文件路径（不用担心用户把软件的名改了）
    exe_path = os.path.abspath(sys.argv[0])
    # 当前目录路径（当前软件文件夹路径）
    directory_path = os.getcwd()
    # 通过环境变量获取Windows临时文件夹路径
    temp_directory = os.environ.get("TEMP") or os.environ.get("TMP")
    # 构建卸载文件bat的路径（放到temp目录中）
    uninstall_bat_path = os.path.join(temp_directory, f"爱丽丝QQ聊天AI{os.getpid()}.bat")
    # 卸载软件的代码
    uninstall_code = f"""
@echo off
ping -n 1000 127.0.0.1
setlocal enabledelayedexpansion
set max_wait=30
set wait_count=0
:check_process
set process_name="爱丽丝QQ聊天Al.exe"
:check_process
tasklist /fi "imagename eq %process_name%" | findstr /i "%process_name%"
if %errorlevel% equ 0 (
    if %wait_count% geq %max_wait% (
        taskkill /f /im "%process_name%"
        exit /b 1
    ) else (
        ping -n 1000 127.0.0.1
        set /a wait_count+=1
        goto :check_process
    )
)
rd /s /q {os.path.join(directory_path,"packages")}
rd /s /q {os.path.join(directory_path,"logs")}
rd /s /q {os.path.join(directory_path,"resources")}
rd /s /q {os.path.join(directory_path,"用户设置")}
rd /s /q {os.path.join(directory_path,"文档")}
del /f /q {exe_path}
pause
del %0
    """
    # 把删除代码写入文件里面去
    with open(uninstall_bat_path, "w", encoding="ANSI") as f:
        f.write(uninstall_code)

    # 启动批处理文件进行删除
    subprocess.Popen(
        [uninstall_bat_path],
        shell=True,                                 # 将通过系统的 shell 程序执行，而不是直接执行
        stdin=subprocess.DEVNULL,                   # 输入重定向为空
        stdout=subprocess.DEVNULL,                  # 标准输出重定向为空
        stderr=subprocess.DEVNULL,                  # 错误出重定向为空
        creationflags=subprocess.CREATE_NO_WINDOW   # 不创建可见窗口（后台运行）
    )
    info(uninstall_bat_path)
```
